#include <Arduino.h>
#include "driver/twai.h"

#define CAN_TX_PIN 21
#define CAN_RX_PIN 22

#define DHT_CAN_ID 0x101

#define TEMP_THRESHOLD 35.0f
#define HUM_THRESHOLD  80.0f


void setup()
{
    Serial.begin(115200);

    delay(1000);


    Serial.println();
    Serial.println("====================================");
    Serial.println("SLAVE 2");
    Serial.println("DHT22");
    Serial.println("CAN ID = 0x101");
    Serial.println("125 kbps");
    Serial.println("====================================");


    /* --------------------------------------------------
       CAN configuration
       -------------------------------------------------- */

    twai_general_config_t g_config =
        TWAI_GENERAL_CONFIG_DEFAULT(
            GPIO_NUM_21,
            GPIO_NUM_22,
            TWAI_MODE_NORMAL
        );


    /* --------------------------------------------------
       125 kbps
       -------------------------------------------------- */

    twai_timing_config_t t_config =
        TWAI_TIMING_CONFIG_125KBITS();


    /* --------------------------------------------------
       Filter ONLY 0x101
       -------------------------------------------------- */

    twai_filter_config_t f_config =
    {
        .acceptance_code =
            ((uint32_t)DHT_CAN_ID << 21),

        .acceptance_mask =
            ~(TWAI_STD_ID_MASK << 21),

        .single_filter = true
    };


    /* --------------------------------------------------
       Install CAN
       -------------------------------------------------- */

    if (twai_driver_install(
            &g_config,
            &t_config,
            &f_config) != ESP_OK)
    {
        Serial.println(
            "CAN DRIVER INSTALL FAILED"
        );

        while (1)
        {
            delay(1000);
        }
    }


    /* --------------------------------------------------
       Start CAN
       -------------------------------------------------- */

    if (twai_start() != ESP_OK)
    {
        Serial.println(
            "CAN START FAILED"
        );

        while (1)
        {
            delay(1000);
        }
    }


    Serial.println(
        "CAN STARTED"
    );

    Serial.println(
        "Waiting for DHT22..."
    );

    Serial.println();
}


void loop()
{
    twai_message_t message;

    if (twai_receive(
            &message,
            pdMS_TO_TICKS(3000)) == ESP_OK)
    {
        if (message.identifier == DHT_CAN_ID &&
            message.data_length_code >= 5)
        {
            Serial.println();
            Serial.println("================================");
            Serial.println("       SLAVE 2 - DHT22");
            Serial.println("================================");

            /*
             * Byte 5 tells us whether PSoC
             * successfully read the DHT22.
             */

            if (message.data[4] == 0xA5)
            {
                int16_t temperature10 =
                    (int16_t)(
                        ((uint16_t)message.data[0] << 8) |
                        message.data[1]
                    );

                uint16_t humidity10 =
                    ((uint16_t)message.data[2] << 8) |
                    message.data[3];

                float temperature =
                    (float)temperature10 / 10.0f;

                float humidity =
                    (float)humidity10 / 10.0f;


                Serial.println("DHT READ : OK");

                Serial.print("CAN ID   : 0x");
                Serial.println(message.identifier, HEX);

                Serial.print("Temperature : ");
                Serial.print(temperature, 1);
                Serial.println(" °C");

                Serial.print("Humidity    : ");
                Serial.print(humidity, 1);
                Serial.println(" % RH");


                /* Temperature threshold */
                if (temperature > 35.0f)
                {
                    Serial.println();
                    Serial.println(
                        "!!! REDUCE TEMPERATURE !!!"
                    );
                }


                /* Humidity threshold */
                if (humidity > 80.0f)
                {
                    Serial.println();
                    Serial.println(
                        "!!! REDUCE HUMIDITY !!!"
                    );


                }

                if (temperature <= 35.0f &&
                    humidity <= 80.0f)
                {
                    Serial.println();
                    Serial.println(
                        "STATUS : NORMAL"
                    );
                }
            }
            else
            {
                Serial.println("DHT READ : FAILED");

                Serial.println(
                    "PSoC could not read the DHT22."
                );
            }

            Serial.println(
                "================================"
            );
        }
    }
}
