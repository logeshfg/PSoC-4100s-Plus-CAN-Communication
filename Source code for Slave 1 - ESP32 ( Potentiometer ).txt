#include <Arduino.h>
#include "driver/twai.h"

#define CAN_TX_PIN 21
#define CAN_RX_PIN 22

#define POT_CAN_ID 0x102

#define POT_THRESHOLD 80.0f


void setup()
{
    Serial.begin(115200);

    delay(1000);

    Serial.println();
    Serial.println("====================================");
    Serial.println("SLAVE 1");
    Serial.println("POTENTIOMETER");
    Serial.println("CAN ID = 0x102");
    Serial.println("125 kbps");
    Serial.println("====================================");


    /* --------------------------------------------------
       General CAN configuration
       -------------------------------------------------- */

    twai_general_config_t g_config =
        TWAI_GENERAL_CONFIG_DEFAULT(
            GPIO_NUM_21,
            GPIO_NUM_22,
            TWAI_MODE_NORMAL
        );


    /* --------------------------------------------------
       125 kbps
       -------------------------------------------------- */

    twai_timing_config_t t_config =
        TWAI_TIMING_CONFIG_125KBITS();


    /* --------------------------------------------------
       Filter ONLY 0x102
       -------------------------------------------------- */

    twai_filter_config_t f_config =
    {
        .acceptance_code =
            ((uint32_t)POT_CAN_ID << 21),

        .acceptance_mask =
            ~(TWAI_STD_ID_MASK << 21),

        .single_filter = true
    };


    /* --------------------------------------------------
       Install CAN
       -------------------------------------------------- */

    if (twai_driver_install(
            &g_config,
            &t_config,
            &f_config) != ESP_OK)
    {
        Serial.println(
            "CAN DRIVER INSTALL FAILED"
        );

        while (1)
        {
            delay(1000);
        }
    }


    /* --------------------------------------------------
       Start CAN
       -------------------------------------------------- */

    if (twai_start() != ESP_OK)
    {
        Serial.println(
            "CAN START FAILED"
        );

        while (1)
        {
            delay(1000);
        }
    }


    Serial.println("CAN STARTED");
    Serial.println(
        "Waiting for potentiometer..."
    );

    Serial.println();
}


void loop()
{
    twai_message_t message;


    if (twai_receive(
            &message,
            pdMS_TO_TICKS(3000)) == ESP_OK)
    {
        if (message.identifier == POT_CAN_ID &&
            message.data_length_code >= 2)
        {
            /* ------------------------------------------
               Convert two CAN bytes back to mV
               ------------------------------------------ */

            uint16_t pot_mV =
                ((uint16_t)message.data[0] << 8) |
                message.data[1];


            /* ------------------------------------------
               Convert voltage to percentage
               ------------------------------------------ */

            float percentage =
                ((float)pot_mV / 3300.0f) *
                100.0f;


            /* Limit */
            if (percentage < 0.0f)
                percentage = 0.0f;

            if (percentage > 100.0f)
                percentage = 100.0f;


            Serial.println();
            Serial.println(
                "================================"
            );

            Serial.println(
                "   SLAVE 1 - POTENTIOMETER"
            );

            Serial.println(
                "================================"
            );


            Serial.print(
                "CAN ID     : 0x"
            );

            Serial.println(
                message.identifier,
                HEX
            );


            Serial.print(
                "Voltage    : "
            );

            Serial.print(
                pot_mV
            );

            Serial.println(
                " mV"
            );


            Serial.print(
                "Percentage : "
            );

            Serial.print(
                percentage,
                1
            );

            Serial.println(
                " %"
            );


            /* ------------------------------------------
               Range
               ------------------------------------------ */

            if (percentage < 25.0f)
            {
                Serial.println(
                    "Range      : LOW"
                );
            }
            else if (percentage < 70.0f)
            {
                Serial.println(
                    "Range      : NORMAL"
                );
            }
            else if (percentage <= POT_THRESHOLD)
            {
                Serial.println(
                    "Range      : HIGH"
                );
            }
            else
            {
                Serial.println(
                    "Range      : VERY HIGH"
                );
            }


            /* ------------------------------------------
               Threshold
               ------------------------------------------ */

            if (percentage > POT_THRESHOLD)
            {
                Serial.println();
                Serial.println(
                    "!!! REDUCE !!!"
                );

                Serial.println(
                    "Potentiometer value is above threshold."
                );
            }
            else
            {
                Serial.println();
                Serial.println(
                    "STATUS     : NORMAL"
                );
            }


            Serial.println(
                "================================"
            );
        }
    }
}
